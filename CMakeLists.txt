cmake_minimum_required(VERSION 3.16)
project(firmware_update_lab LANGUAGES CXX)
enable_testing()
add_executable(firmware_tests tests/test.cpp)
target_include_directories(firmware_tests PRIVATE include)
set_target_properties(firmware_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(firmware_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME firmware_tests COMMAND firmware_tests)
